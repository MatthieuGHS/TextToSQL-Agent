# === Modèle et orchestration ===
anthropic                    # SDK Anthropic
langchain-anthropic          # ChatAnthropic : connecteur Claude pour LangChain
langchain-core               # types de base (messages, outils)
#
# `langgraph` figurait ici et n'y figure plus : la boucle d'E4 est écrite à la main, et
# aucun module ne l'importe. Une dépendance non utilisée est une ligne de maintenance et
# une surface d'attaque pour rien.

# === Données ===
duckdb                       # moteur SQL analytique local
pandas                       # manipulation des résultats
openpyxl                     # lecture de la grille d'évaluation .xlsx

# === Interface ===
fastapi                      # E9 — API HTTP, coquille mince autour de `ask()`
uvicorn[standard]            # serveur ASGI
python-multipart             # téléversement des fichiers sources (formulaire multipart)
#
# `streamlit` et `plotly` étaient prévus ici. L'interface est finalement React + Vite
# (voir `web/`), et le rendu des graphiques d'E7 se fera côté navigateur à partir d'une
# spécification produite par `src/charts/` — donc sans dépendance Python de rendu.

# === Outils ===
python-dotenv                # chargement du .env
pytest                       # E5 — harnais d'évaluation
